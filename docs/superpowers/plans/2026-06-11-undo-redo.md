# Undo / Redo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Undo/Redo buttons to the overlay panel backed by an agent-side in-memory global stack: one Apply = one undoable step; Undo restores the pre-Apply file, Redo re-applies.

**Architecture:** A pure `history.ts` stack in the agent records `{file, before, after}` per applied edit; `/undo`, `/redo`, `/history` HTTP routes drive file writes; the overlay panel gains two buttons that call them and reflect availability.

**Tech Stack:** TypeScript, Node http (agent), ts-morph (existing edit path), esbuild (overlay bundle), vitest.

**Spec:** `docs/superpowers/specs/2026-06-11-undo-redo-design.md`

**Conventions:** Run tests `npx vitest run`; typecheck `npx tsc --noEmit`; build overlay `npm run build:overlay`. Agent tests in `tests/agent/`, overlay has no DOM unit tests (verified by browser smoke). Commit after each task.

```ts
// src/agent/history.ts

interface EditRecord { file: string; before: string; after: string; }

export interface History {
  /** Record an applied edit; clears the redo stack. */
  record(file: string, before: string, after: string): void;
  /** Pop the newest undo record; returns the content to write (before), or null. */
  undo(): { file: string; content: string } | null;
  /** Pop the newest redo record; returns the content to write (after), or null. */
  redo(): { file: string; content: string } | null;
  state(): { canUndo: boolean; canRedo: boolean };
}

/** In-memory global undo/redo stack for applied edits. */
export function createHistory(): History {
  const undoStack: EditRecord[] = [];
  const redoStack: EditRecord[] = [];
  return {
    record(file, before, after) {
      undoStack.push({ file, before, after });
      redoStack.length = 0;
    },
    undo() {
      const r = undoStack.pop();
      if (!r) return null;
      redoStack.push(r);
      return { file: r.file, content: r.before };
    },
    redo() {
      const r = redoStack.pop();
      if (!r) return null;
      undoStack.push(r);
      return { file: r.file, content: r.after };
    },
    state() {
      return { canUndo: undoStack.length > 0, canRedo: redoStack.length > 0 };
    },
  };
}
```

```ts
    $<HTMLButtonElement>("undo").disabled = !canUndo;
    $<HTMLButtonElement>("redo").disabled = !canRedo;
  }

  async function runHistory(kind: "undo" | "redo") {
    try {
      const r = kind === "undo" ? await handlers.onUndo() : await handlers.onRedo();
      if (r.status === "error") { out.textContent = `❌ ${r.message}`; return; }
      setHistoryButtons(r.canUndo, r.canRedo);
      if (r.status === "ok") {
        const short = r.file.split(/[\\/]/).pop();
        out.textContent = kind === "undo" ? `↶ ${short} 되돌림` : `↷ ${short} 다시 적용`;
        // Refresh rows only if the affected file is the one currently shown.
        if (r.file === file) await inspectInto();
      }
    } catch (e) {
      out.textContent = `❌ agent unreachable: ${(e as Error).message}`;
    }

  $("redo").onclick = () => void runHistory("redo");
```
